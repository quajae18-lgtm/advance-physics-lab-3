// Set up the pins
// Our digital PWN output is set to Digital Out 9
const int outPin = 9;

// Start with an output of 0 and declare this variable as an integer
int outputLevel=0;

void setup() {
// Define the pin mode for the digital output
pinMode(outPin, OUTPUT);
Serial.begin(9600);
Serial.println("Time,Heatcontrollervalue,Thermistorvalue");
}
void loop() {
// Make an oscillating value
int Thermistorvalue = analogRead(A0);
int Heatcontrollervalue = analogRead(A1);

Serial.print(millis());
Serial.print(",");

Serial.print(Heatcontrollervalue);
Serial.print(",");

Serial.println(Thermistorvalue);

outputLevel = map(Heatcontrollervalue,0,1023,0,255);


// Write the outputLevel variable to our digital output pin
analogWrite(outPin, outputLevel);

delay(200);
}
